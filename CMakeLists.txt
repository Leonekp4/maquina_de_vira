cmake_minimum_required(VERSION 3.16)
project(maquina_vira LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(controller firmware/controle-esp32/src/controller.cpp)
target_include_directories(controller PUBLIC firmware/controle-esp32/include)
target_compile_options(controller PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(simulador firmware/controle-esp32/sim/main.cpp)
target_link_libraries(simulador PRIVATE controller)
add_executable(controller_tests firmware/controle-esp32/test/controller_tests.cpp)
target_link_libraries(controller_tests PRIVATE controller)
add_test(NAME controller_tests COMMAND controller_tests)
add_executable(ihm_tests firmware/ihm-esp32s3/test/ui_values_tests.cpp)
target_include_directories(ihm_tests PRIVATE firmware/ihm-esp32s3/include)
target_link_libraries(ihm_tests PRIVATE controller)
add_test(NAME ihm_tests COMMAND ihm_tests)
add_executable(ihm_session_tests firmware/ihm-esp32s3/test/ui_session_tests.cpp)
target_include_directories(ihm_session_tests PRIVATE firmware/ihm-esp32s3/include)
target_link_libraries(ihm_session_tests PRIVATE controller)
add_test(NAME ihm_session_tests COMMAND ihm_session_tests)
